public void updateReadingBankDataZero(ActionEvent event) {
    String msg = ResourceBundle.getBundle("/MyBundle").getString("Item" + "Updated");

    try {
        progress = BigDecimal.ZERO;
        BigDecimal progressRate = BigDecimal.ONE;
        if (!selectedReadingNew2List.isEmpty()) {
            progressRate = new BigDecimal(100)
                .divide(new BigDecimal(selectedReadingNew2List.size()), 2, RoundingMode.CEILING);
        }

        for (BillingReading read : selectedReadingNew2List) {

            // 1) (Re)format phone (not really used in the update payload)
            String phone = read.getCustomerInfoId().getPhoneNumber();
            String formatedphone = "0900000000";
            if (phone != null && phone.length() == 13) {
                formatedphone = phone.replace("+251", "0");
            } else if (phone != null && phone.length() == 10) {
                formatedphone = phone.replace("+251", "0");
            }
            if (formatedphone != null && !formatedphone.matches("^[0-9]{10}$")) {
                formatedphone = "0900000000";
            }

            CompanyProfile cp = this.ejbFacade.findAllCompanyProfileLatest();

            // 2) Build DerashCustomerBillDataUpdate payload
            DerashCustomerBillDataUpdate dinfo = new DerashCustomerBillDataUpdate();

            dinfo.setAmount_due("0");  // zero bill at Derash
            dinfo.setBill_desc("Previous:0, Penality:0, Current:0, Total:0");
            dinfo.setBill_id(read.getInvoiceNumberId().getInvoiceNumbers());
            dinfo.setDue_date(
                new SimpleDateFormat("yyyy-MM-dd")
                    .format(getAddDaytoJdkDate(getCurrentDateJdk(), -1))
            );
            dinfo.setReason(
                getMonthNameFromAmharicMName(getMonthNameFromKfyawor(read.getKifyaWer()))
                + "," + getYearFromKfyawor(read.getKifyaWer())
            );

            // 3) If templateDerashMessage is set, override desc+reason with template text
            if (cp.getTemplateDerashMessage() != null) {
                String message = cp.getTemplateDerashMessage();
                String[] kfyamonth = read.getKifyaWer().split(",");

                message = message.replaceAll("#10", read.getLastReading() + "");
                message = message.replaceAll("#1",  read.getCustomerInfoId().getFullNameEng());
                message = message.replaceAll("#2",
                    getMonthNameFromAmharicMName(kfyamonth[0].trim()) + ", " + kfyamonth[1].trim());
                message = message.replaceAll("#3",  read.getConsumption() + "");
                message = message.replaceAll("#4",  read.getWuzifKezihEske() != null ? read.getWuzifKezihEske() : "");
                message = message.replaceAll("#5", (read.getWuzifHisab() + read.getWuzifKotariKiray()) + "");
                message = message.replaceAll("#6",  read.getTekilalaTekefay() + "");
                message = message.replaceAll("#7",  read.getCustomerInfoId().getAccountNumber() + "");
                message = message.replaceAll("#8",  getInEthiopian(read.getBankDueDate()) + "");
                message = message.replaceAll("#9",  read.getPreviousReading() + "");

                dinfo.setBill_desc(message);
                dinfo.setReason(message);
            }

            // 4) Call Derash API to update the bill
            companyProfile = this.ejbFacade.findAllCompanyProfileLatest();
            DerashCustomerBillInfoFacade studentRectClient = new DerashCustomerBillInfoFacade(companyProfile);
            DerashSendBill derash = studentRectClient.updateData(dinfo);

            if (derash != null) {
                // 5) Locally mark bill as canceled
                read.setIsBankCanceled(true);
                this.ejbFacade.editReading(read);
            }
            progress = progress.add(progressRate);
        }

        progress = new BigDecimal(100);

        JsfUtil.addSuccessMessage("ባንክ ተልኳል", "ወደ ባንክ በትክክል ተልኳል።");
        setSelectedCustomerInfoList(null);
        setReadingList(null);

    } catch (Exception ex) {
        ...
    }
}
